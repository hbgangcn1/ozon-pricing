# 微信支付配置示例
# 请设置以下环境变量或在代码中直接配置

# 微信公众号/小程序 AppID
WECHAT_APP_ID=your_app_id_here

# 微信支付商户号
WECHAT_MCH_ID=your_merchant_id_here

# 微信支付API密钥
WECHAT_API_KEY=your_api_key_here

# 商户证书路径（可选，用于退款等需要证书的接口）
WECHAT_CERT_PATH=path/to/your/cert.pem
WECHAT_KEY_PATH=path/to/your/key.pem

# 支付回调通知地址
WECHAT_NOTIFY_URL=https://your-domain.com/wechat/notify

# 支付完成跳转地址
WECHAT_RETURN_URL=https://your-domain.com/payment/success

# 是否使用沙箱环境（测试用）
WECHAT_SANDBOX=false
